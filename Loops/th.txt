1. for loop
    for(initialization; condition; increment/decrement) {
        // code block to be executed
    }

2. while loop
    let initialization;
    while(condition) {
        // code block to be executed
        increment/decrement;
    }

3. do-while loop
    let initialization;
    do {
        // code block to be executed
        increment/decrement;
    }while(condition);
